SELECT *
FROM (
  SELECT T.RMA_NO, T.OBJID, T.OBJVERSION
    FROM IFSAPP.RETURN_MATERIAL T
   WHERE T.CUST_REF               = 'ITICOG0124103535'
     AND T.RETURN_APPROVER_ID     = 'HOPUR20'
     AND T.CUSTOMER_NO            = 'S00443'
     AND T.CONTRACT               = 'COVAL'
     AND T.STATE                  = 'Planned'
     AND TRUNC(T.DATE_REQUESTED)  = TRUNC(SYSDATE)
     AND T.RMA_NO NOT IN          (SELECT R.rma_no FROM IFSAPP.RETURN_MATERIAL_LINE R WHERE R.debit_invoice_no =
T.CUST_REF)

   ORDER BY T.OBJVERSION DESC
)
WHERE ROWNUM = 1;

14. Fetch Oldest Inventory for Reservation

SELECT T.lot_batch_no,
       T.waiv_dev_rej_no,
       SUM(T.qty_onhand - T.qty_reserved) AS Available_Quantity
FROM IFSAPP.INVENTORY_PART_IN_STOCK T
WHERE T.contract  = :site
  AND T.part_no   = :part_no
  AND SUBSTR(T.location_no, 1, INSTR(T.location_no, '-GST') - 1) = :location_group
  AND (T.qty_onhand - T.qty_reserved) > 0
GROUP BY T.lot_batch_no, T.waiv_dev_rej_no, T.receipt_date
HAVING SUM(T.qty_onhand - T.qty_reserved) > 0
ORDER BY T.receipt_date ASC;
